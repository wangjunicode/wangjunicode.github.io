# 游戏运行时场景编辑器设计与实现完全指南

## 前言

游戏运行时场景编辑器（Runtime Scene Editor）是一类让玩家或关卡设计师在**游戏运行状态下**直接编辑关卡、布置物体、调整场景元素的工具系统。区别于 Unity Editor 本身的离线编辑器，Runtime Editor 运行在游戏进程中，支持实时预览效果、热部署编辑结果，广泛应用于：

- **建造类游戏**：《我的世界》、《围攻》等玩家自建世界
- **地图编辑器**：《星际争霸》、《魔兽争霸》等 UGC 关卡工具
- **关卡设计辅助工具**：策划/TA 在游戏内实时调整关卡，无需重新打包
- **沙盒类游戏**：房间装饰、基地建设系统

本文从零设计一个工业级 Runtime Scene Editor，覆盖架构设计、核心模块实现、序列化存储与最佳实践。

## 核心模块一：命令模式与撤销重做

### IEditorCommand 接口

```csharp
/// <summary>
/// 编辑器命令接口，所有编辑操作必须实现此接口
/// </summary>
public interface IEditorCommand
{
    string CommandName { get; }
    void Execute();
    void Undo();
    // 合并相邻同类命令（如连续移动只记录一条）
    bool TryMerge(IEditorCommand other);
}
```

### CommandManager：撤销栈管理

```csharp
public class CommandManager
{
    private readonly Stack<IEditorCommand> _undoStack = new Stack<IEditorCommand>();
    private readonly Stack<IEditorCommand> _redoStack = new Stack<IEditorCommand>();
    private const int MaxHistorySize = 200;

    public event Action OnHistoryChanged;

    public void Execute(IEditorCommand command)
    {
        // 尝试与上一条命令合并（例如连续微调位置）
        if (_undoStack.Count > 0 && _undoStack.Peek().TryMerge(command))
        {
            // 合并成功，更新已有命令的状态但不推新命令
            OnHistoryChanged?.Invoke();
            return;
        }

        command.Execute();
        _undoStack.Push(command);
        _redoStack.Clear(); // 执行新操作时清空重做栈

        // 限制历史记录大小
        if (_undoStack.Count > MaxHistorySize)
        {
            var arr = _undoStack.ToArray();
            _undoStack.Clear();
            for (int i = MaxHistorySize - 1; i >= 0; i--)
                _undoStack.Push(arr[i]);
        }

        OnHistoryChanged?.Invoke();
    }

    public void Undo()
    {
        if (_undoStack.Count == 0) return;
        var cmd = _undoStack.Pop();
        cmd.Undo();
        _redoStack.Push(cmd);
        OnHistoryChanged?.Invoke();
    }

    public void Redo()
    {
        if (_redoStack.Count == 0) return;
        var cmd = _redoStack.Pop();
        cmd.Execute();
        _undoStack.Push(cmd);
        OnHistoryChanged?.Invoke();
    }

    public bool CanUndo => _undoStack.Count > 0;
    public bool CanRedo => _redoStack.Count > 0;
    public string CurrentUndoLabel => _undoStack.Count > 0 ? _undoStack.Peek().CommandName : string.Empty;
}
```

### 具体命令示例：移动命令

```csharp
public class MoveCommand : IEditorCommand
{
    private readonly List<TransformSnapshot> _snapshots;
    private readonly Vector3 _delta;
    private Vector3 _accumulatedDelta; // 用于合并

    public string CommandName => "移动对象";

    public MoveCommand(IEnumerable<Transform> targets, Vector3 delta)
    {
        _delta = delta;
        _accumulatedDelta = delta;
        _snapshots = targets.Select(t => new TransformSnapshot(t)).ToList();
    }

    public void Execute()
    {
        foreach (var snap in _snapshots)
            snap.Target.position += _accumulatedDelta;
    }

    public void Undo()
    {
        foreach (var snap in _snapshots)
            snap.Target.position = snap.OriginalPosition;
    }

    public bool TryMerge(IEditorCommand other)
    {
        if (other is MoveCommand move && IsSameTargets(move))
        {
            _accumulatedDelta += move._delta;
            // 直接应用追加的 delta，不需要再 Execute
            foreach (var snap in _snapshots)
                snap.Target.position += move._delta;
            return true;
        }
        return false;
    }

    private bool IsSameTargets(MoveCommand other)
    {
        if (_snapshots.Count != other._snapshots.Count) return false;
        for (int i = 0; i < _snapshots.Count; i++)
            if (_snapshots[i].Target != other._snapshots[i].Target) return false;
        return true;
    }
}

public struct TransformSnapshot
{
    public Transform Target;
    public Vector3 OriginalPosition;
    public Quaternion OriginalRotation;
    public Vector3 OriginalScale;

    public TransformSnapshot(Transform t)
    {
        Target = t;
        OriginalPosition = t.position;
        OriginalRotation = t.rotation;
        OriginalScale = t.localScale;
    }
}
```

## 核心模块二：Transform Gizmo 操控器

### Gizmo 状态机

```csharp
public enum GizmoMode { None, Translate, Rotate, Scale }
public enum GizmoAxis { None, X, Y, Z, XY, XZ, YZ, XYZ }

public class TransformGizmo : MonoBehaviour
{
    [Header("Settings")]
    public float GizmoSize = 1.5f;
    public float AxisHandleThickness = 0.08f;
    public float RotationRingRadius = 1.2f;

    [Header("Colors")]
    public Color AxisColorX = new Color(1f, 0.2f, 0.2f);
    public Color AxisColorY = new Color(0.2f, 1f, 0.2f);
    public Color AxisColorZ = new Color(0.2f, 0.4f, 1f);
    public Color HighlightColor = Color.yellow;

    private GizmoMode _mode = GizmoMode.Translate;
    private GizmoAxis _hoveredAxis = GizmoAxis.None;
    private GizmoAxis _draggingAxis = GizmoAxis.None;
    private Vector3 _dragStartPoint;
    private List<TransformSnapshot> _dragSnapshots = new();
    private SelectionManager _selection;
    private CommandManager _commandManager;
    private Camera _camera;

    private void Start()
    {
        _selection = FindObjectOfType<SelectionManager>();
        _commandManager = FindObjectOfType<CommandManager>();
        _camera = Camera.main;
    }

    private void Update()
    {
        if (_selection.SelectedObjects.Count == 0) return;

        Vector3 center = GetSelectionCenter();
        float screenScale = GetScreenScaleFactor(center);

        HandleHover(center, screenScale);
        HandleDrag(center);
    }

    private Vector3 GetSelectionCenter()
    {
        if (_selection.SelectedObjects.Count == 0) return Vector3.zero;
        Vector3 sum = Vector3.zero;
        foreach (var go in _selection.SelectedObjects)
            sum += go.transform.position;
        return sum / _selection.SelectedObjects.Count;
    }

    // 根据相机距离计算 Gizmo 屏幕恒定大小
    private float GetScreenScaleFactor(Vector3 worldPos)
    {
        float dist = Vector3.Distance(_camera.transform.position, worldPos);
        return dist * GizmoSize * 0.1f;
    }

    private void HandleHover(Vector3 center, float scale)
    {
        Ray ray = _camera.ScreenPointToRay(Input.mousePosition);
        _hoveredAxis = HitTestAxes(ray, center, scale);
    }

    private GizmoAxis HitTestAxes(Ray ray, Vector3 center, float scale)
    {
        // 对每个轴做射线-胶囊体碰撞检测
        float minDist = float.MaxValue;
        GizmoAxis result = GizmoAxis.None;

        var axes = new[]
        {
            (GizmoAxis.X, Vector3.right),
            (GizmoAxis.Y, Vector3.up),
            (GizmoAxis.Z, Vector3.forward)
        };

        foreach (var (axis, dir) in axes)
        {
            Vector3 axisEnd = center + dir * scale;
            float dist = RaySegmentDistance(ray, center, axisEnd);
            if (dist < AxisHandleThickness * scale && dist < minDist)
            {
                minDist = dist;
                result = axis;
            }
        }

        return result;
    }

    private float RaySegmentDistance(Ray ray, Vector3 segStart, Vector3 segEnd)
    {
        // 计算射线与线段之间的最近距离
        Vector3 d1 = ray.direction.normalized;
        Vector3 d2 = (segEnd - segStart).normalized;
        Vector3 r = ray.origin - segStart;

        float a = Vector3.Dot(d1, d1);
        float e = Vector3.Dot(d2, d2);
        float f = Vector3.Dot(d2, r);
        float c = Vector3.Dot(d1, r);
        float b = Vector3.Dot(d1, d2);
        float denom = a * e - b * b;

        float s = Mathf.Clamp((b * f - c * e) / denom, 0f, 1f);
        float t = Mathf.Clamp((a * f - b * c) / denom, 0f, Vector3.Distance(segStart, segEnd));

        Vector3 p1 = ray.origin + d1 * s;
        Vector3 p2 = segStart + d2 * t;
        return Vector3.Distance(p1, p2);
    }

    private void HandleDrag(Vector3 center)
    {
        if (Input.GetMouseButtonDown(0) && _hoveredAxis != GizmoAxis.None)
        {
            _draggingAxis = _hoveredAxis;
            _dragStartPoint = GetAxisProjectionPoint(center);
            _dragSnapshots = _selection.SelectedObjects
                .Select(go => new TransformSnapshot(go.transform))
                .ToList();
        }

        if (Input.GetMouseButton(0) && _draggingAxis != GizmoAxis.None)
        {
            Vector3 currentPoint = GetAxisProjectionPoint(center);
            Vector3 delta = currentPoint - _dragStartPoint;
            ApplyDelta(delta);
            _dragStartPoint = currentPoint;
        }

        if (Input.GetMouseButtonUp(0) && _draggingAxis != GizmoAxis.None)
        {
            // 拖拽结束，提交命令到撤销栈
            var targets = _selection.SelectedObjects.Select(go => go.transform);
            // 实际上移动已经实时应用，这里提交一个记录快照的命令
            _commandManager.Execute(new FinalizeMoveCommand(_dragSnapshots));
            _draggingAxis = GizmoAxis.None;
        }
    }

    private Vector3 GetAxisProjectionPoint(Vector3 center)
    {
        Ray ray = _camera.ScreenPointToRay(Input.mousePosition);
        Vector3 axisDir = _draggingAxis switch
        {
            GizmoAxis.X => Vector3.right,
            GizmoAxis.Y => Vector3.up,
            GizmoAxis.Z => Vector3.forward,
            _ => Vector3.up
        };

        // 将鼠标射线投影到轴线上
        Vector3 normal = Vector3.Cross(axisDir, _camera.transform.forward).normalized;
        normal = Vector3.Cross(normal, axisDir).normalized;
        Plane plane = new Plane(normal, center);

        if (plane.Raycast(ray, out float enter))
        {
            Vector3 hitPoint = ray.GetPoint(enter);
            // 约束到轴方向
            float projection = Vector3.Dot(hitPoint - center, axisDir);
            return center + axisDir * projection;
        }
        return center;
    }

    private void ApplyDelta(Vector3 delta)
    {
        foreach (var go in _selection.SelectedObjects)
            go.transform.position += delta;
    }
}
```

## 核心模块三：多选与框选系统

```csharp
public class SelectionManager : MonoBehaviour
{
    public List<GameObject> SelectedObjects { get; private set; } = new();
    public event Action<List<GameObject>> OnSelectionChanged;

    [Header("Box Selection")]
    [SerializeField] private RectTransform _boxSelectionRect;
    private Vector2 _boxStartPos;
    private bool _isBoxSelecting;
    private Camera _camera;

    // 可选对象层级标记
    [SerializeField] private LayerMask _selectableLayers;

    private void Start()
    {
        _camera = Camera.main;
    }

    private void Update()
    {
        HandleClickSelection();
        HandleBoxSelection();
    }

    private void HandleClickSelection()
    {
        if (Input.GetMouseButtonDown(0) && !_isBoxSelecting)
        {
            Ray ray = _camera.ScreenPointToRay(Input.mousePosition);
            if (Physics.Raycast(ray, out RaycastHit hit, 1000f, _selectableLayers))
            {
                var go = hit.collider.gameObject;
                if (Input.GetKey(KeyCode.LeftShift))
                {
                    // Shift 追加/取消选择
                    if (SelectedObjects.Contains(go))
                        SelectedObjects.Remove(go);
                    else
                        SelectedObjects.Add(go);
                }
                else if (Input.GetKey(KeyCode.LeftControl))
                {
                    // Ctrl 多选
                    if (!SelectedObjects.Contains(go))
                        SelectedObjects.Add(go);
                }
                else
                {
                    SelectedObjects.Clear();
                    SelectedObjects.Add(go);
                }
                OnSelectionChanged?.Invoke(SelectedObjects);
            }
            else if (!Input.GetKey(KeyCode.LeftShift) && !Input.GetKey(KeyCode.LeftControl))
            {
                SelectedObjects.Clear();
                OnSelectionChanged?.Invoke(SelectedObjects);
            }
        }
    }

    private void HandleBoxSelection()
    {
        // 中键或右键拖拽触发框选
        if (Input.GetMouseButtonDown(1))
        {
            _boxStartPos = Input.mousePosition;
            _isBoxSelecting = true;
            _boxSelectionRect.gameObject.SetActive(true);
        }

        if (_isBoxSelecting && Input.GetMouseButton(1))
        {
            UpdateBoxSelectionVisual();
        }

        if (_isBoxSelecting && Input.GetMouseButtonUp(1))
        {
            _isBoxSelecting = false;
            _boxSelectionRect.gameObject.SetActive(false);
            ExecuteBoxSelect();
        }
    }

    private void UpdateBoxSelectionVisual()
    {
        Vector2 current = Input.mousePosition;
        Vector2 lowerLeft = new Vector2(
            Mathf.Min(_boxStartPos.x, current.x),
            Mathf.Min(_boxStartPos.y, current.y));
        Vector2 size = new Vector2(
            Mathf.Abs(current.x - _boxStartPos.x),
            Mathf.Abs(current.y - _boxStartPos.y));

        _boxSelectionRect.anchoredPosition = lowerLeft;
        _boxSelectionRect.sizeDelta = size;
    }

    private void ExecuteBoxSelect()
    {
        Vector2 current = Input.mousePosition;
        Rect screenRect = new Rect(
            Mathf.Min(_boxStartPos.x, current.x),
            Mathf.Min(_boxStartPos.y, current.y),
            Mathf.Abs(current.x - _boxStartPos.x),
            Mathf.Abs(current.y - _boxStartPos.y));

        if (!Input.GetKey(KeyCode.LeftShift))
            SelectedObjects.Clear();

        foreach (var selectable in FindObjectsOfType<SelectableObject>())
        {
            Vector3 screenPos = _camera.WorldToScreenPoint(selectable.transform.position);
            if (screenRect.Contains(new Vector2(screenPos.x, screenPos.y)))
            {
                if (!SelectedObjects.Contains(selectable.gameObject))
                    SelectedObjects.Add(selectable.gameObject);
            }
        }
        OnSelectionChanged?.Invoke(SelectedObjects);
    }

    public void SelectAll()
    {
        SelectedObjects = FindObjectsOfType<SelectableObject>()
            .Select(s => s.gameObject).ToList();
        OnSelectionChanged?.Invoke(SelectedObjects);
    }

    public void ClearSelection()
    {
        SelectedObjects.Clear();
        OnSelectionChanged?.Invoke(SelectedObjects);
    }
}

/// <summary>
/// 挂载到可被选中的对象上
/// </summary>
public class SelectableObject : MonoBehaviour { }
```

## 核心模块四：对齐与吸附系统

```csharp
public class SnapSystem : MonoBehaviour
{
    [Header("Grid Snap")]
    public bool GridSnapEnabled = true;
    public float GridSize = 0.5f;

    [Header("Angle Snap")]
    public bool AngleSnapEnabled = true;
    public float AngleStep = 15f;

    [Header("Surface Snap")]
    public bool SurfaceSnapEnabled = false;
    public LayerMask SurfaceLayer;

    [Header("Vertex Snap")]
    public bool VertexSnapEnabled = false;
    public float VertexSnapRadius = 0.3f;

    public Vector3 SnapPosition(Vector3 worldPos)
    {
        if (VertexSnapEnabled)
        {
            var snapped = TryVertexSnap(worldPos);
            if (snapped.HasValue) return snapped.Value;
        }

        if (SurfaceSnapEnabled)
        {
            var snapped = TrySurfaceSnap(worldPos);
            if (snapped.HasValue) return snapped.Value;
        }

        if (GridSnapEnabled)
            return SnapToGrid(worldPos);

        return worldPos;
    }

    public Quaternion SnapRotation(Quaternion rotation)
    {
        if (!AngleSnapEnabled) return rotation;

        Vector3 euler = rotation.eulerAngles;
        euler.x = Mathf.Round(euler.x / AngleStep) * AngleStep;
        euler.y = Mathf.Round(euler.y / AngleStep) * AngleStep;
        euler.z = Mathf.Round(euler.z / AngleStep) * AngleStep;
        return Quaternion.Euler(euler);
    }

    private Vector3 SnapToGrid(Vector3 pos)
    {
        return new Vector3(
            Mathf.Round(pos.x / GridSize) * GridSize,
            Mathf.Round(pos.y / GridSize) * GridSize,
            Mathf.Round(pos.z / GridSize) * GridSize);
    }

    private Vector3? TrySurfaceSnap(Vector3 pos)
    {
        if (Physics.Raycast(pos + Vector3.up * 5f, Vector3.down, out RaycastHit hit, 20f, SurfaceLayer))
            return hit.point;
        return null;
    }

    private Vector3? TryVertexSnap(Vector3 pos)
    {
        // 寻找附近最近的顶点
        Collider[] nearbyColliders = Physics.OverlapSphere(pos, VertexSnapRadius);
        float minDist = float.MaxValue;
        Vector3? closest = null;

        foreach (var col in nearbyColliders)
        {
            if (col.TryGetComponent<MeshFilter>(out var mf) && mf.sharedMesh != null)
            {
                foreach (var v in mf.sharedMesh.vertices)
                {
                    Vector3 worldV = col.transform.TransformPoint(v);
                    float dist = Vector3.Distance(pos, worldV);
                    if (dist < minDist)
                    {
                        minDist = dist;
                        closest = worldV;
                    }
                }
            }
        }
        return closest;
    }
}
```

```csharp
[System.Serializable]
public class SceneData

    public string PrefabPath;    // Resources 路径或 Addressable key
    public SerializableVector3 Position;
    public SerializableQuaternion Rotation;
    public SerializableVector3 Scale;
    public string ParentId;      // 父对象 ID，支持嵌套
    public Dictionary<string, string> CustomProperties = new(); // 扩展属性
}

[System.Serializable]
public struct SerializableVector3
{
    public float x, y, z;
    public static implicit operator Vector3(SerializableVector3 v) => new(v.x, v.y, v.z);
    public static implicit operator SerializableVector3(Vector3 v) => new() { x = v.x, y = v.y, z = v.z };
}

[System.Serializable]
public struct SerializableQuaternion
{

    public static implicit operator Quaternion(SerializableQuaternion q) => new(q.x, q.y, q.z, q.w);
    public static implicit operator SerializableQuaternion(Quaternion q) => new() { x = q.x, y = q.y, z = q.z, w = q.w };
}

public class SceneSerializer : MonoBehaviour
{
    private const string SaveDirectory = "UserScenes";
    
    // 将场景内所有可编辑对象序列化并保存
    public void SaveScene(string sceneName)
    {
        var data = new SceneData
        {
            SceneName = sceneName,
            SaveTimestamp = DateTimeOffset.UtcNow.ToUnixTimeSeconds()
        };

        var objects = FindObjectsOfType<EditableObject>();
        foreach (var obj in objects)
        {
            var objData = new SceneObjectData
            {
                InstanceId = obj.InstanceId,
                PrefabPath = obj.PrefabPath,
                Position = obj.transform.position,
                Rotation = obj.transform.rotation,
                Scale = obj.transform.localScale,
                ParentId = obj.transform.parent?.GetComponent<EditableObject>()?.InstanceId
            };
            obj.SerializeCustomProperties(objData.CustomProperties);
            data.Objects.Add(objData);
        }

        string json = JsonUtility.ToJson(data, true);
        string path = Path.Combine(Application.persistentDataPath, SaveDirectory, $"{sceneName}.json");
        Directory.CreateDirectory(Path.GetDirectoryName(path)!);
        File.WriteAllText(path, json);
        
        Debug.Log($"[SceneSerializer] 场景已保存: {path}，共 {data.Objects.Count} 个对象");
    }

    // 加载场景数据并重建场景对象
    public async UniTask LoadScene(string sceneName)
    {
        string path = Path.Combine(Application.persistentDataPath, SaveDirectory, $"{sceneName}.json");
        if (!File.Exists(path))
        {
            Debug.LogWarning($"[SceneSerializer] 场景文件不存在: {path}");
            return;
        }

        // 清除当前场景所有可编辑对象
        foreach (var obj in FindObjectsOfType<EditableObject>())
            Destroy(obj.gameObject);

        string json = await File.ReadAllTextAsync(path);
        var data = JsonUtility.FromJson<SceneData>(json);

        // 第一遍：创建所有对象
        var instanceMap = new Dictionary<string, EditableObject>();
        foreach (var objData in data.Objects)
        {
            var prefab = await Addressables.LoadAssetAsync<GameObject>(objData.PrefabPath).Task;
            var go = Instantiate(prefab);
            var editable = go.GetOrAddComponent<EditableObject>();
            editable.InstanceId = objData.InstanceId;
            editable.PrefabPath = objData.PrefabPath;
            go.transform.position = objData.Position;
            go.transform.rotation = objData.Rotation;
            go.transform.localScale = objData.Scale;
            editable.DeserializeCustomProperties(objData.CustomProperties);
            instanceMap[objData.InstanceId] = editable;
        }

        // 第二遍：重建父子关系
        foreach (var objData in data.Objects)
        {
            if (!string.IsNullOrEmpty(objData.ParentId) &&
                instanceMap.TryGetValue(objData.ParentId, out var parent) &&
                instanceMap.TryGetValue(objData.InstanceId, out var child))
            {
                child.transform.SetParent(parent.transform, true);
            }
        }

        Debug.Log($"[SceneSerializer] 场景加载完成: {sceneName}，共 {data.Objects.Count} 个对象");
    }
}

/// <summary>
/// 可编辑对象组件，附加在所有可被 RuntimeEditor 操作的 GameObject 上
/// </summary>
public class EditableObject : MonoBehaviour
{
    [SerializeField] private string _instanceId;
    [SerializeField] private string _prefabPath;

    public string InstanceId
    {
        get
        {
            if (string.IsNullOrEmpty(_instanceId))
                _instanceId = Guid.NewGuid().ToString();
            return _instanceId;
        }
        set => _instanceId = value;
    }

    public string PrefabPath
    {
        get => _prefabPath;
```

```csharp
public class ObjectPlacer : MonoBehaviour
{
    [SerializeField] private Material _previewMaterial;
    [SerializeField] private SnapSystem _snapSystem;
    [SerializeField] private CommandManager _commandManager;

    private GameObject _previewObject;
    private string _currentPrefabPath;
    private Camera _camera;
    private bool _isPlacing;

    private void Start()
    {
        _camera = Camera.main;
    }

    public void BeginPlacement(string prefabPath, GameObject prefab)
    {
        _currentPrefabPath = prefabPath;
        _isPlacing = true;

        // 创建半透明预览对象
        _previewObject = Instantiate(prefab);
        SetPreviewMaterial(_previewObject, _previewMaterial);
        // 禁用碰撞，防止干扰射线检测
        foreach (var col in _previewObject.GetComponentsInChildren<Collider>())
            col.enabled = false;
    }

    public void CancelPlacement()
    {
        _isPlacing = false;
        if (_previewObject != null)
        {
            Destroy(_previewObject);
            _previewObject = null;
        }
    }

    private void Update()
    {
        if (!_isPlacing || _previewObject == null) return;

        // 更新预览位置
        Ray ray = _camera.ScreenPointToRay(Input.mousePosition);
        if (Physics.Raycast(ray, out RaycastHit hit, 1000f))
        {
            Vector3 snapped = _snapSystem.SnapPosition(hit.point);
            _previewObject.transform.position = snapped;
        }

        // 旋转预览（Q/E键）
        if (Input.GetKey(KeyCode.Q))
            _previewObject.transform.Rotate(Vector3.up, -45f * Time.deltaTime);
        if (Input.GetKey(KeyCode.E))
            _previewObject.transform.Rotate(Vector3.up, 45f * Time.deltaTime);

        // 确认放置
        if (Input.GetMouseButtonDown(0))
        {
            PlaceObject();
        }

        // 取消放置
        if (Input.GetKeyDown(KeyCode.Escape))
        {
            CancelPlacement();
        }
    }

    private void PlaceObject()
    {
        var cmd = new PlaceObjectCommand(_currentPrefabPath,
            _previewObject.transform.position,
            _previewObject.transform.rotation,
            _previewObject.transform.localScale);
        _commandManager.Execute(cmd);
    }

    private void SetPreviewMaterial(GameObject go, Material mat)
    {
        foreach (var renderer in go.GetComponentsInChildren<Renderer>())
        {
            var mats = new Material[renderer.materials.Length];
            for (int i = 0; i < mats.Length; i++)
                mats[i] = mat;
            renderer.materials = mats;
        }
    }
}

public class PlaceObjectCommand : IEditorCommand
{
    private readonly string _prefabPath;
    private readonly Vector3 _position;
    private readonly Quaternion _rotation;
    private readonly Vector3 _scale;
    private GameObject _placedObject;

    public string CommandName => "放置对象";

    public PlaceObjectCommand(string prefabPath, Vector3 pos, Quaternion rot, Vector3 scale)
    {
        _prefabPath = prefabPath;
        _position = pos;
        _rotation = rot;
        _scale = scale;
    }

    public async void Execute()
    {
        var prefab = await Addressables.LoadAssetAsync<GameObject>(_prefabPath).Task;
        _placedObject = Object.Instantiate(prefab, _position, _rotation);
        _placedObject.transform.localScale = _scale;
        var editable = _placedObject.GetOrAddComponent<EditableObject>();
        editable.PrefabPath = _prefabPath;
    }

    public void Undo()
    {
        if (_placedObject != null)
            Object.Destroy(_placedObject);
    }

    public bool TryMerge(IEditorCommand other) => false;
}
```

```csharp
/// <summary>

[RequireComponent(typeof(Camera))]
public class GizmoRenderer : MonoBehaviour
{
    [SerializeField] private TransformGizmo _gizmo;
    [SerializeField] private SelectionManager _selection;
    [SerializeField] private Material _lineMaterial;

    private void OnPostRender()
    {
        if (_selection.SelectedObjects.Count == 0) return;

        GL.PushMatrix();
        _lineMaterial.SetPass(0);
        GL.Begin(GL.LINES);

        // 绘制选择边界框
        foreach (var go in _selection.SelectedObjects)
            DrawBoundingBox(go);

        // 绘制 Gizmo 轴线
        DrawTransformGizmo();

        // 绘制网格辅助线
        DrawGridLines();

        GL.End();
        GL.PopMatrix();
    }

    private void DrawBoundingBox(GameObject go)
    {
        var bounds = CalculateBounds(go);
        Vector3 min = bounds.min;
        Vector3 max = bounds.max;

        GL.Color(Color.white);

        // 底面
        DrawLine(new Vector3(min.x, min.y, min.z), new Vector3(max.x, min.y, min.z));
        DrawLine(new Vector3(max.x, min.y, min.z), new Vector3(max.x, min.y, max.z));
        DrawLine(new Vector3(max.x, min.y, max.z), new Vector3(min.x, min.y, max.z));
        DrawLine(new Vector3(min.x, min.y, max.z), new Vector3(min.x, min.y, min.z));
        // 顶面
        DrawLine(new Vector3(min.x, max.y, min.z), new Vector3(max.x, max.y, min.z));
        DrawLine(new Vector3(max.x, max.y, min.z), new Vector3(max.x, max.y, max.z));
        DrawLine(new Vector3(max.x, max.y, max.z), new Vector3(min.x, max.y, max.z));
        DrawLine(new Vector3(min.x, max.y, max.z), new Vector3(min.x, max.y, min.z));
        // 竖边
        DrawLine(new Vector3(min.x, min.y, min.z), new Vector3(min.x, max.y, min.z));
        DrawLine(new Vector3(max.x, min.y, min.z), new Vector3(max.x, max.y, min.z));
        DrawLine(new Vector3(max.x, min.y, max.z), new Vector3(max.x, max.y, max.z));
        DrawLine(new Vector3(min.x, min.y, max.z), new Vector3(min.x, max.y, max.z));
    }

    private void DrawTransformGizmo()
    {
        // 由 TransformGizmo 组件负责提供中心点
        // 此处绘制三色坐标轴
    }

    private void DrawGridLines()
    {
        int count = 20;
        float size = 10f;
        GL.Color(new Color(0.3f, 0.3f, 0.3f, 0.5f));
        for (int i = -count; i <= count; i++)
        {
            float pos = i * (size / count);
            DrawLine(new Vector3(-size, 0, pos), new Vector3(size, 0, pos));
            DrawLine(new Vector3(pos, 0, -size), new Vector3(pos, 0, size));
        }
    }

    private void DrawLine(Vector3 from, Vector3 to)
    {
        GL.Vertex(from);
        GL.Vertex(to);
    }

    private Bounds CalculateBounds(GameObject go)
    {
        var renderers = go.GetComponentsInChildren<Renderer>();
        if (renderers.Length == 0) return new Bounds(go.transform.position, Vector3.one * 0.5f);
        var bounds = renderers[0].bounds;
        for (int i = 1; i < renderers.Length; i++)
            bounds.Encapsulate(renderers[i].bounds);
        return bounds;
    }
```

```csharp
        if (Input.GetKeyDown(KeyCode.E)) _gizmo.SetMode(GizmoMode.Rotate);

    }

    private void DeleteSelected()
    {
        if (_selection.SelectedObjects.Count == 0) return;
        var cmd = new DeleteObjectsCommand(_selection.SelectedObjects.ToList());
        _commandManager.Execute(cmd);
        _selection.ClearSelection();
    }

    private void DuplicateSelected()
    {
        if (_selection.SelectedObjects.Count == 0) return;
        var cmd = new DuplicateObjectsCommand(_selection.SelectedObjects.ToList());
        _commandManager.Execute(cmd);
    }

    {
        if (_selection.SelectedObjects.Count == 0) return;

            .Aggregate(Vector3.zero, (acc, go) => acc + go.transform.position)
            / _selection.SelectedObjects.Count;
        // 平滑移动相机到选中对象
        Camera.main.transform.DOMove(center + Camera.main.transform.forward * -5f, 0.3f);
```

```csharp
public class SelectionQuadTree
{
    private QuadTreeNode _root;

    public void Rebuild(IEnumerable<EditableObject> objects, Bounds worldBounds)
    {
        _root = new QuadTreeNode(worldBounds, 4);
        foreach (var obj in objects)
            _root.Insert(obj);
    }

    public List<EditableObject> QueryRect(Rect screenRect, Camera camera)
    {
        // 将屏幕矩形转为世界 AABB 后查询四叉树
        var worldBounds = ScreenRectToWorldBounds(screenRect, camera);
        return _root.Query(worldBounds);
    }
}
```

## 最佳实践总结

| 实践点 | 说明 |
|--------|------|
| **命令模式是核心** | 所有编辑操作必须封装为 IEditorCommand，确保撤销/重做可靠 |
| **合并相邻命令** | 连续微调位置/旋转应合并为单条命令，避免历史栈爆炸 |
| **预览对象独立层** | 放置预览对象应放到独立 Layer，避免影响正常物理/射线检测 |
| **延迟自动保存** | 使用 Dirty 标记 + 定时器，避免每帧序列化 |
| **InstanceId 持久化** | 每个可编辑对象必须有 GUID，保证保存/加载时引用稳定 |
| **分离编辑与游戏逻辑** | Editor 代码与游戏逻辑用 Layer/Assembly 隔离，Release 包可裁剪 |
| **空间分区加速选取** | 使用四叉树/八叉树加速框选，场景规模大时性能差距显著 |
| **Gizmo LOD** | 相机距离远时简化 Gizmo 渲染，节省 Draw Call |

---

## 总结

一个工业级 Runtime Scene Editor 的核心在于：

1. **命令模式** 保证所有操作可撤销，是系统可靠性的基础
2. **Transform Gizmo** 是交互体验的关键，需精心处理轴线投影和拖拽合并
3. **序列化系统** 决定编辑结果的持久化质量，需考虑版本兼容性
4. **对齐吸附** 大幅提升关卡设计效率，网格/顶点/表面三种模式缺一不可
5. **性能优化** 在复杂场景下尤为重要，Dirty 标记、四叉树、Gizmo LOD 是常用手段

掌握这套技术体系，可以为游戏添加完整的 UGC 关卡编辑功能，或为团队打造高效的内置关卡设计工具。
