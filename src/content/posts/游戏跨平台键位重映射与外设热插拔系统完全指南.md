# 游戏跨平台键位重映射与外设热插拔系统完全指南

现代游戏的输入系统已远超"读取按键"的范畴：玩家期望能够**自定义键位**、随时**切换输入设备**、在手柄断连后**无缝继续游戏**，而开发者则需要同时管理 PC 键鼠、多型号手柄、移动端触屏乃至主机平台的差异。

本文基于 **Unity New Input System**，系统性地构建一套完整的键位重映射与外设管理框架，覆盖从底层事件到上层 UI 的全链路实现。

## 2. 外设热插拔检测系统

### 2.1 设备类型枚举与识别

```csharp
using UnityEngine;
using UnityEngine.InputSystem;
using System;
using System.Collections.Generic;

/// <summary>
/// 外设类型枚举
/// </summary>
public enum DeviceType
{
    Unknown,
    KeyboardMouse,
    Gamepad_Xbox,
    Gamepad_PlayStation,
    Gamepad_Switch,
    Gamepad_Generic,
    Touchscreen,
    Joystick
}

/// <summary>
/// 外设热插拔管理器
/// </summary>
public class DeviceManager : MonoBehaviour
{
    public static DeviceManager Instance { get; private set; }

    // 当前活跃设备类型
    public DeviceType CurrentDeviceType { get; private set; } = DeviceType.KeyboardMouse;

    // 事件：设备类型改变时触发
    public event Action<DeviceType, DeviceType> OnDeviceTypeChanged;
    // 事件：具体设备接入
    public event Action<InputDevice> OnDeviceConnected;
    // 事件：具体设备断开
    public event Action<InputDevice> OnDeviceDisconnected;

    // 已连接的手柄列表
    public IReadOnlyList<Gamepad> ConnectedGamepads => _connectedGamepads;
    private readonly List<Gamepad> _connectedGamepads = new();

    void Awake()
    {
        if (Instance != null) { Destroy(gameObject); return; }
        Instance = this;
        DontDestroyOnLoad(gameObject);

        // 注册设备变化回调
        InputSystem.onDeviceChange += OnInputDeviceChange;

        // 注册"任意动作"回调，用于检测当前活跃设备
        InputSystem.onActionChange += OnAnyActionChange;

        // 初始化已连接设备列表
        RefreshConnectedGamepads();
    }

    void OnDestroy()
    {
        InputSystem.onDeviceChange -= OnInputDeviceChange;
        InputSystem.onActionChange -= OnAnyActionChange;
    }

    /// <summary>
    /// 设备连接/断开回调
    /// </summary>
    private void OnInputDeviceChange(InputDevice device, InputDeviceChange change)
    {
        switch (change)
        {
            case InputDeviceChange.Added:
                Debug.Log($"[DeviceManager] 设备接入: {device.displayName} ({device.GetType().Name})");
                RefreshConnectedGamepads();
                OnDeviceConnected?.Invoke(device);

                // 手柄接入时自动切换到手柄模式
                if (device is Gamepad)
                    SwitchToDevice(IdentifyGamepadType(device as Gamepad));
                break;

            case InputDeviceChange.Removed:
                Debug.Log($"[DeviceManager] 设备断开: {device.displayName}");
                RefreshConnectedGamepads();
                OnDeviceDisconnected?.Invoke(device);

                // 手柄全部断开时回退到键鼠
                if (device is Gamepad && _connectedGamepads.Count == 0)
                    SwitchToDevice(DeviceType.KeyboardMouse);
                break;

            case InputDeviceChange.Reconnected:
                RefreshConnectedGamepads();
                OnDeviceConnected?.Invoke(device);
                break;
        }
    }

    /// <summary>
    /// 通过动作触发检测当前活跃输入设备（自动切换 UI 图标提示）
    /// </summary>
    private void OnAnyActionChange(object actionOrMap, InputActionChange change)
    {
        if (change != InputActionChange.ActionPerformed) return;

        var action = actionOrMap as InputAction;
        if (action == null) return;

        var device = action.activeControl?.device;
        if (device == null) return;

        DeviceType newType = IdentifyDeviceType(device);
        if (newType != CurrentDeviceType)
            SwitchToDevice(newType);
    }

    private void SwitchToDevice(DeviceType newType)
    {
        if (newType == CurrentDeviceType) return;
        var oldType = CurrentDeviceType;
        CurrentDeviceType = newType;
        Debug.Log($"[DeviceManager] 活跃设备切换: {oldType} → {newType}");
        OnDeviceTypeChanged?.Invoke(oldType, newType);
    }

    private DeviceType IdentifyDeviceType(InputDevice device)
    {
        if (device is Keyboard || device is Mouse) return DeviceType.KeyboardMouse;
        if (device is Touchscreen)                 return DeviceType.Touchscreen;
        if (device is Gamepad gp)                  return IdentifyGamepadType(gp);
        if (device is Joystick)                    return DeviceType.Joystick;
        return DeviceType.Unknown;
    }

    private DeviceType IdentifyGamepadType(Gamepad gp)
    {
        string name = gp.displayName.ToLowerInvariant();
        if (name.Contains("xbox") || name.Contains("xinput"))       return DeviceType.Gamepad_Xbox;
        if (name.Contains("dualshock") || name.Contains("dualsense")
            || name.Contains("playstation") || name.Contains("ps4")
            || name.Contains("ps5"))                                 return DeviceType.Gamepad_PlayStation;
        if (name.Contains("switch") || name.Contains("pro controller")
            || name.Contains("joycon"))                              return DeviceType.Gamepad_Switch;
        return DeviceType.Gamepad_Generic;
    }

    private void RefreshConnectedGamepads()
    {
        _connectedGamepads.Clear();
        foreach (var gp in Gamepad.all)
            _connectedGamepads.Add(gp);
    }

    /// <summary>
    /// 获取当前主手柄（如有多个取第一个）
    /// </summary>
    public Gamepad GetPrimaryGamepad()
        => _connectedGamepads.Count > 0 ? _connectedGamepads[0] : null;
}
```

## 3. 键位重映射核心系统

### 3.1 重映射管理器

```csharp
using UnityEngine;
using UnityEngine.InputSystem;
using System;
using System.Collections;
using System.Collections.Generic;

/// <summary>
/// 重映射操作状态
/// </summary>
public enum RebindState
{
    Idle,
    WaitingForInput,
    Completed,
    Canceled,
    Conflict
}

/// <summary>
/// 键位重映射管理器
/// </summary>
public class InputRebindManager : MonoBehaviour
{
    public static InputRebindManager Instance { get; private set; }

    [Header("Input Action Asset")]
    public InputActionAsset actionAsset;

    // 重映射操作状态
    public RebindState CurrentState { get; private set; } = RebindState.Idle;

    // 事件
    public event Action<InputAction, int, RebindState> OnRebindStateChanged;
    public event Action<string> OnRebindConflict;   // 参数为冲突的动作名称

    // 当前进行中的重映射操作
    private InputActionRebindingExtensions.RebindingOperation _currentOperation;

    // 键位覆盖存储键名
    private const string SAVE_KEY = "InputBindingOverrides_v2";

    void Awake()
    {
        if (Instance != null) { Destroy(gameObject); return; }
        Instance = this;
        DontDestroyOnLoad(gameObject);
        LoadBindings();
    }

    void OnDestroy()
    {
        _currentOperation?.Dispose();
    }

    /// <summary>
    /// 开始监听重映射输入
    /// </summary>
    /// <param name="action">要重映射的动作</param>
    /// <param name="bindingIndex">绑定索引（-1 表示第一个匹配当前设备的绑定）</param>
    /// <param name="onComplete">完成回调</param>
    public void StartRebinding(
        InputAction action,
        int         bindingIndex,
        Action<bool> onComplete = null)
    {
        if (CurrentState == RebindState.WaitingForInput)
        {
            CancelRebinding();
        }

        // 禁用动作，避免自身触发
        action.Disable();

        CurrentState = RebindState.WaitingForInput;
        OnRebindStateChanged?.Invoke(action, bindingIndex, CurrentState);

        _currentOperation = action
            .PerformInteractiveRebinding(bindingIndex)
            .WithControlsExcluding("<Mouse>/position")  // 排除鼠标位移
            .WithControlsExcluding("<Mouse>/delta")
            .WithControlsExcluding("<Gamepad>/Start")   // 排除系统按键
            .WithCancelingThrough("<Keyboard>/escape")  // Esc 取消
            .OnMatchWaitForAnother(0.1f)                // 等待 0.1s 确保不是组合键
            .OnComplete(op =>
            {
                string conflictAction = CheckBindingConflict(action, bindingIndex);
                if (conflictAction != null)
                {
                    // 检测到冲突
                    CurrentState = RebindState.Conflict;
                    OnRebindStateChanged?.Invoke(action, bindingIndex, CurrentState);
                    OnRebindConflict?.Invoke(conflictAction);
                    // 询问玩家是否强制覆盖（由 UI 层处理）
                }
                else
                {
                    CurrentState = RebindState.Completed;
                    OnRebindStateChanged?.Invoke(action, bindingIndex, CurrentState);
                }

                _currentOperation.Dispose();
                _currentOperation = null;
                action.Enable();
                SaveBindings();

                onComplete?.Invoke(conflictAction == null);
            })
            .OnCancel(op =>
            {
                CurrentState = RebindState.Canceled;
                OnRebindStateChanged?.Invoke(action, bindingIndex, CurrentState);
                _currentOperation.Dispose();
                _currentOperation = null;
                action.Enable();
                onComplete?.Invoke(false);
            })
            .Start();
    }

    /// <summary>
    /// 取消当前重映射
    /// </summary>
    public void CancelRebinding()
    {
        _currentOperation?.Cancel();
    }

    /// <summary>
    /// 重置单个动作的键位为默认
    /// </summary>
    public void ResetBinding(InputAction action, int bindingIndex)
    {
        action.RemoveBindingOverride(bindingIndex);
        SaveBindings();
    }

    /// <summary>
    /// 重置所有键位为默认
    /// </summary>
    public void ResetAllBindings()
    {
        foreach (var map in actionAsset.actionMaps)
            map.RemoveAllBindingOverrides();
        PlayerPrefs.DeleteKey(SAVE_KEY);
        PlayerPrefs.Save();
        Debug.Log("[InputRebindManager] 所有键位已重置为默认");
    }

    /// <summary>
    /// 获取绑定的显示名称（用于 UI 显示）
    /// </summary>
    public string GetBindingDisplayName(InputAction action, int bindingIndex)
    {
        return action.GetBindingDisplayString(
            bindingIndex,
            InputBinding.DisplayStringOptions.DontIncludeInteractions);
    }

    /// <summary>
    /// 检测键位冲突（同一 ActionMap 内）
    /// </summary>
    private string CheckBindingConflict(InputAction targetAction, int bindingIndex)
    {
        var targetBinding = targetAction.bindings[bindingIndex];
        var targetMap     = targetAction.actionMap;

        foreach (var action in targetMap.actions)
        {
            if (action == targetAction) continue;
            for (int i = 0; i < action.bindings.Count; i++)
            {
                var binding = action.bindings[i];
                // 比较有效路径
                string effective = string.IsNullOrEmpty(binding.overridePath)
                    ? binding.path
                    : binding.overridePath;
                string targetEffective = string.IsNullOrEmpty(targetBinding.overridePath)
                    ? targetBinding.path
                    : targetBinding.overridePath;

                if (effective == targetEffective && !string.IsNullOrEmpty(effective))
                    return action.name;
            }
        }
        return null;
    }

    /// <summary>
    /// 持久化存储键位覆盖
    /// </summary>
    public void SaveBindings()
    {
        string json = actionAsset.SaveBindingOverridesAsJson();
        PlayerPrefs.SetString(SAVE_KEY, json);
        PlayerPrefs.Save();
        Debug.Log($"[InputRebindManager] 键位已保存（{json.Length} bytes）");
    }

    /// <summary>
    /// 加载持久化键位覆盖
    /// </summary>
    public void LoadBindings()
    {
        if (!PlayerPrefs.HasKey(SAVE_KEY)) return;
        string json = PlayerPrefs.GetString(SAVE_KEY);
        if (string.IsNullOrEmpty(json)) return;

        try
        {
            actionAsset.LoadBindingOverridesFromJson(json);
            Debug.Log("[InputRebindManager] 键位已从存档加载");
        }
        catch (Exception e)
        {
            Debug.LogWarning($"[InputRebindManager] 键位加载失败，使用默认键位: {e.Message}");
            PlayerPrefs.DeleteKey(SAVE_KEY);
        }
    }

    /// <summary>
    /// 根据绑定 id 定位绑定索引（含 Composite 支持）
    /// </summary>
    public int GetBindingIndexForControl(InputAction action, string controlScheme)
    {
        for (int i = 0; i < action.bindings.Count; i++)
        {
            var binding = action.bindings[i];
            if (binding.isComposite) continue;
            if (binding.groups.Contains(controlScheme))
                return i;
        }
        return 0;
    }
}
```

### 3.2 重映射 UI 面板

```csharp
using UnityEngine;
using UnityEngine.UI;
using TMPro;
using UnityEngine.InputSystem;

/// <summary>
/// 单个键位绑定行 UI 组件
/// </summary>
public class RebindActionRow : MonoBehaviour
{
    [Header("UI References")]
    public TextMeshProUGUI actionNameLabel;
    public Button          rebindButton;
    public TextMeshProUGUI currentBindingLabel;
    public Button          resetButton;
    public GameObject      waitingOverlay;   // 等待输入时显示的遮罩

    private InputAction _action;
    private int         _bindingIndex;

    public void Initialize(InputAction action, int bindingIndex)
    {
        _action       = action;
        _bindingIndex = bindingIndex;

        actionNameLabel.text = action.name;
        RefreshBindingDisplay();

        rebindButton.onClick.AddListener(StartRebind);
        resetButton.onClick.AddListener(ResetBinding);

        // 监听重映射状态变化
        InputRebindManager.Instance.OnRebindStateChanged += OnRebindStateChanged;
    }

    void OnDestroy()
    {
        if (InputRebindManager.Instance != null)
            InputRebindManager.Instance.OnRebindStateChanged -= OnRebindStateChanged;
    }

    private void StartRebind()
    {
        InputRebindManager.Instance.StartRebinding(_action, _bindingIndex, success =>
        {
            RefreshBindingDisplay();
        });
    }

    private void ResetBinding()
    {
        InputRebindManager.Instance.ResetBinding(_action, _bindingIndex);
        RefreshBindingDisplay();
    }

    private void RefreshBindingDisplay()
    {
        currentBindingLabel.text = InputRebindManager.Instance
            .GetBindingDisplayName(_action, _bindingIndex);
    }

    private void OnRebindStateChanged(InputAction action, int index, RebindState state)
    {
        if (action != _action || index != _bindingIndex) return;

        bool isWaiting = state == RebindState.WaitingForInput;
        waitingOverlay.SetActive(isWaiting);
        rebindButton.interactable = !isWaiting;

        if (state == RebindState.Completed)
        {
            currentBindingLabel.text = InputRebindManager.Instance
                .GetBindingDisplayName(_action, _bindingIndex);
        }
    }
}
```

## 4. 平台差异与 Control Scheme 适配

### 4.1 Control Scheme 自动切换

```csharp
using UnityEngine.InputSystem;
using UnityEngine;

/// <summary>
/// 根据当前活跃设备自动切换 Control Scheme
/// </summary>
public class ControlSchemeManager : MonoBehaviour
{
    [Header("Player Input")]
    public PlayerInput playerInput;

    // Control Scheme 名称常量（需与 Input Action Asset 中定义一致）
    private const string SCHEME_KBM       = "Keyboard&Mouse";
    private const string SCHEME_GAMEPAD   = "Gamepad";
    private const string SCHEME_TOUCH     = "Touch";

    void Start()
    {
        DeviceManager.Instance.OnDeviceTypeChanged += HandleDeviceTypeChanged;
    }

    void OnDestroy()
    {
        if (DeviceManager.Instance != null)
            DeviceManager.Instance.OnDeviceTypeChanged -= HandleDeviceTypeChanged;
    }

    private void HandleDeviceTypeChanged(DeviceType oldType, DeviceType newType)
    {
        string schemeName = newType switch
        {
            DeviceType.KeyboardMouse    => SCHEME_KBM,
            DeviceType.Gamepad_Xbox
                or DeviceType.Gamepad_PlayStation
                or DeviceType.Gamepad_Switch
                or DeviceType.Gamepad_Generic => SCHEME_GAMEPAD,
            DeviceType.Touchscreen      => SCHEME_TOUCH,
            _                           => SCHEME_KBM
        };

        if (playerInput.currentControlScheme != schemeName)
        {
            // 切换设备
            var device = GetPreferredDevice(newType);
            if (device != null)
                playerInput.SwitchCurrentControlScheme(schemeName, device);
            Debug.Log($"[ControlSchemeManager] 切换 Scheme: {schemeName}");
        }
    }

    private InputDevice GetPreferredDevice(DeviceType type)
    {
        return type switch
        {
            DeviceType.KeyboardMouse    => Keyboard.current,
            DeviceType.Touchscreen      => Touchscreen.current,
            _                           => DeviceManager.Instance.GetPrimaryGamepad()
        };
    }
}
```

### 4.2 设备图标动态替换系统

```csharp
using UnityEngine;
using UnityEngine.UI;
using System.Collections.Generic;

/// <summary>
/// 根据当前活跃设备动态替换 UI 上的按键图标
/// </summary>
public class DeviceIconReplacer : MonoBehaviour
{
    [System.Serializable]
    public struct IconEntry
    {
        public string      actionName;    // 对应的 InputAction 名称
        public Sprite      xboxIcon;
        public Sprite      psIcon;
        public Sprite      switchIcon;
        public Sprite      kbmIcon;
        public Sprite      touchIcon;
    }

    [Header("图标配置")]
    public List<IconEntry> iconEntries;

    [Header("UI 图标目标（与 iconEntries 对应）")]
    public List<Image> targetImages;

    void Start()
    {
        DeviceManager.Instance.OnDeviceTypeChanged += _ => RefreshAllIcons();
        RefreshAllIcons();
    }

    void OnDestroy()
    {
        if (DeviceManager.Instance != null)
            DeviceManager.Instance.OnDeviceTypeChanged -= _ => RefreshAllIcons();
    }

    private void RefreshAllIcons()
    {
        var deviceType = DeviceManager.Instance.CurrentDeviceType;
        for (int i = 0; i < iconEntries.Count && i < targetImages.Count; i++)
        {
            var entry  = iconEntries[i];
            var sprite = deviceType switch
            {
                DeviceType.Gamepad_Xbox        => entry.xboxIcon,
                DeviceType.Gamepad_PlayStation => entry.psIcon,
                DeviceType.Gamepad_Switch      => entry.switchIcon,
                DeviceType.Touchscreen         => entry.touchIcon,
                _                              => entry.kbmIcon
            };

            if (targetImages[i] != null && sprite != null)
                targetImages[i].sprite = sprite;
        }
    }
}
```

## 5. 断线重连与降级处理

### 5.1 手柄断连后的游戏暂停与恢复

```csharp
using UnityEngine;
using UnityEngine.InputSystem;
using System.Collections;

/// <summary>
/// 手柄断连处理器：断连时暂停游戏，重连后恢复
/// </summary>
public class GamepadDisconnectHandler : MonoBehaviour
{
    [Header("断连 UI")]
    public GameObject disconnectPanel;   // 提示"请重新连接手柄"的 UI

    [Header("Settings")]
    [Tooltip("仅在手柄模式下触发暂停")]
    public bool pauseOnlyInGamepadMode = true;

    [Tooltip("等待重连超时时间（秒），-1 表示无限等待")]
    public float reconnectTimeout = 60f;

    private bool _wasGamepadMode;
    private Coroutine _timeoutCoroutine;

    void Start()
    {
        DeviceManager.Instance.OnDeviceDisconnected += HandleDisconnected;
        DeviceManager.Instance.OnDeviceConnected    += HandleConnected;
        if (disconnectPanel) disconnectPanel.SetActive(false);
    }

    void OnDestroy()
    {
        if (DeviceManager.Instance == null) return;
        DeviceManager.Instance.OnDeviceDisconnected -= HandleDisconnected;
        DeviceManager.Instance.OnDeviceConnected    -= HandleConnected;
    }

    private void HandleDisconnected(InputDevice device)
    {
        if (!(device is Gamepad)) return;
        if (DeviceManager.Instance.ConnectedGamepads.Count > 0) return;  // 还有其他手柄

        if (pauseOnlyInGamepadMode
            && DeviceManager.Instance.CurrentDeviceType == DeviceType.KeyboardMouse)
            return;  // 当前是键鼠模式，不需要暂停

        Debug.Log("[GamepadDisconnectHandler] 手柄断连，暂停游戏");
        _wasGamepadMode = true;
        Time.timeScale = 0f;
        if (disconnectPanel) disconnectPanel.SetActive(true);

        if (reconnectTimeout > 0)
        {
            _timeoutCoroutine = StartCoroutine(ReconnectTimeoutCoroutine());
        }
    }

    private void HandleConnected(InputDevice device)
    {
        if (!_wasGamepadMode) return;
        if (!(device is Gamepad)) return;

        Debug.Log("[GamepadDisconnectHandler] 手柄重连，恢复游戏");
        _wasGamepadMode = false;
        Time.timeScale = 1f;

        if (disconnectPanel) disconnectPanel.SetActive(false);
        if (_timeoutCoroutine != null)
        {
            StopCoroutine(_timeoutCoroutine);
            _timeoutCoroutine = null;
        }
    }

    private IEnumerator ReconnectTimeoutCoroutine()
    {
        yield return new WaitForSecondsRealtime(reconnectTimeout);
        if (_wasGamepadMode)
        {
            // 超时未重连：自动降级到键鼠模式，恢复游戏
            Debug.Log("[GamepadDisconnectHandler] 重连超时，降级到键鼠模式");
            _wasGamepadMode = false;
            Time.timeScale = 1f;
            if (disconnectPanel) disconnectPanel.SetActive(false);
        }
    }
}
```

### 6.1 分屏玩家绑定器

```csharp
using UnityEngine;
using UnityEngine.InputSystem;
using System.Collections.Generic;

/// <summary>
/// 本地多人分屏输入管理：每个玩家绑定到独立的输入设备
/// </summary>
public class LocalMultiplayerInputManager : MonoBehaviour
{
    public static LocalMultiplayerInputManager Instance { get; private set; }

    [Header("Input Action Asset")]
    public InputActionAsset actionAsset;

    [Header("最大玩家数")]
    public int maxPlayers = 4;

    public struct PlayerBinding
    {
        public int         playerId;
        public InputDevice device;
        public DeviceType  deviceType;
    }

    private readonly List<PlayerBinding> _playerBindings = new();
    public IReadOnlyList<PlayerBinding> PlayerBindings => _playerBindings;

    // 事件：玩家加入/退出
    public event System.Action<int, InputDevice> OnPlayerJoined;
    public event System.Action<int>              OnPlayerLeft;

    void Awake()
    {
        if (Instance != null) { Destroy(gameObject); return; }
        Instance = this;
    }

    /// <summary>
    /// 注册玩家到指定设备
    /// </summary>
    public bool RegisterPlayer(int playerId, InputDevice device)
    {
        if (_playerBindings.Count >= maxPlayers)
        {
            Debug.LogWarning($"[MultiplayerInput] 玩家数量已满（{maxPlayers}）");
            return false;
        }

        // 检查该设备是否已被占用
        foreach (var binding in _playerBindings)
        {
            if (binding.device == device)
            {
                Debug.LogWarning($"[MultiplayerInput] 设备 {device.displayName} 已被玩家 {binding.playerId} 占用");
                return false;
            }
        }

        _playerBindings.Add(new PlayerBinding
        {
            playerId   = playerId,
            device     = device,
            deviceType = GetDeviceType(device)
        });

        Debug.Log($"[MultiplayerInput] 玩家 {playerId} 绑定到设备: {device.displayName}");
        OnPlayerJoined?.Invoke(playerId, device);
        return true;
    }

    /// <summary>
    /// 注销玩家
    /// </summary>
    public void UnregisterPlayer(int playerId)
    {
        for (int i = _playerBindings.Count - 1; i >= 0; i--)
        {
            if (_playerBindings[i].playerId == playerId)
            {
                _playerBindings.RemoveAt(i);
                Debug.Log($"[MultiplayerInput] 玩家 {playerId} 已注销");
                OnPlayerLeft?.Invoke(playerId);
                return;
            }
        }
    }

    private DeviceType GetDeviceType(InputDevice device)
    {
        if (device is Keyboard || device is Mouse) return DeviceType.KeyboardMouse;
        if (device is Gamepad)                     return DeviceType.Gamepad_Generic;
        if (device is Touchscreen)                 return DeviceType.Touchscreen;
        return DeviceType.Unknown;
    }
}
```

## 7. 键位存档版本迁移

```csharp
/// <summary>
/// 键位存档版本迁移器
/// 当游戏更新后 InputAction 结构改变，旧存档可能失效，需要做迁移或清空
/// </summary>
public static class BindingSaveMigrator
{
    private const string VERSION_KEY = "InputBindings_SchemaVersion";
    private const int    CURRENT_VERSION = 2;

    public static void MigrateIfNeeded(InputActionAsset actionAsset)
    {
        int savedVersion = PlayerPrefs.GetInt(VERSION_KEY, 0);

        if (savedVersion == CURRENT_VERSION)
            return;  // 版本一致，无需迁移

        Debug.Log($"[BindingMigrator] 键位存档版本不匹配: saved={savedVersion}, current={CURRENT_VERSION}");

        if (savedVersion == 0)
        {
            // 版本 0 → 版本 2：全量清空（首次安装或超旧版本）
            PlayerPrefs.DeleteKey("InputBindingOverrides_v1");
            PlayerPrefs.DeleteKey("InputBindingOverrides_v2");
            Debug.Log("[BindingMigrator] 旧键位存档已清空，使用默认键位");
        }
        else if (savedVersion == 1)
        {
            // 版本 1 → 版本 2：尝试迁移，失败则清空
            string oldJson = PlayerPrefs.GetString("InputBindingOverrides_v1", "");
            if (!string.IsNullOrEmpty(oldJson))
            {
                try
                {
                    actionAsset.LoadBindingOverridesFromJson(oldJson);
                    // 迁移成功后保存到新键
                    PlayerPrefs.SetString("InputBindingOverrides_v2",
                        actionAsset.SaveBindingOverridesAsJson());
                    Debug.Log("[BindingMigrator] 键位存档从 v1 迁移到 v2 成功");
                }
                catch
                {
                    Debug.LogWarning("[BindingMigrator] 键位存档迁移失败，清空并使用默认键位");
                    PlayerPrefs.DeleteKey("InputBindingOverrides_v1");
                }
            }
        }

        PlayerPrefs.SetInt(VERSION_KEY, CURRENT_VERSION);
        PlayerPrefs.Save();
    }
}
```

## 8. 最佳实践总结

| 问题 | 推荐方案 |
|------|----------|
| 多设备图标自动切换 | 监听 `onActionChange` 检测活跃设备，驱动 UI 图标切换 |
| 手柄断连后的游戏处理 | 立即暂停 + 显示提示UI + 超时后降级到键鼠 |
| 重映射键位持久化 | `SaveBindingOverridesAsJson` + PlayerPrefs（带版本号） |
| 同一动作多个绑定（如手柄+键鼠各有不同按键） | 使用 Control Scheme 分组，各 Scheme 独立绑定，互不干扰 |
| 复合动作（WASD方向）重映射 | 逐个子绑定（Up/Down/Left/Right）分别 Rebind |
| 避免系统键误绑 | `WithControlsExcluding` 排除 Escape、系统键等 |
| 多人本地分屏 | `PlayerInput` + `PlayerInputManager` 配合独立 `InputActionAsset` 实例 |

**工程核心经验：**

1. **设备切换要区分"硬件事件"与"用户意图"** —— 硬件连接是事件（onDeviceChange），而"当前活跃设备"应从实际触发的动作中推断（onActionChange），二者要分开处理。
2. **Control Scheme 是隔离不同设备键位的关键** —— 同一个动作在 Gamepad Scheme 下绑 A键、在 KBM Scheme 下绑 Space，互不干扰，切换 Scheme 即切换键位集。
3. **重映射必须禁用动作** —— 在 `PerformInteractiveRebinding` 前必须 `action.Disable()`，否则重映射输入会同时触发原动作。
4. **键位冲突检测要给用户选择** —— 检测到冲突时不要静默处理，弹出 UI 让玩家决定是否强制替换，这是良好 UX 的关键。
5. **存档要带版本号** —— 游戏迭代后 ActionAsset 结构可能变化，版本号迁移机制可以避免旧存档导致的崩溃。
